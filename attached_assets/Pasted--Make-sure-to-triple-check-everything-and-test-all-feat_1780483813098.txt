    cache = redis.Redis(host='localhost', port=6379, db=0, decode_responses=True)
except Exception as e:
    logging.error(f"Redis initialization failed: {e}")
    cache = None

def get_cached_football_data(cache_key, db_fetch_callback, expiry_seconds=3600):
    """
    Generic wrapper to cache static technique videos or position workout templates.
    """
    if cache:
        try:
            cached_data = cache.get(cache_key)
            if cached_data:
                return json.loads(cached_data)  # Instant return
        except Exception as e:
            logging.warning(f"Cache read error: {e}")

    # Fallback to main database fetch
    fresh_data = db_fetch_callback()

    if cache and fresh_data:
        try:
            cache.setex(cache_key, expiry_seconds, json.dumps(fresh_data))
        except Exception as e:
            logging.warning(f"Cache write error: {e}")

    return fresh_data

async function routeCacheMiddleware(req, res, next) {
    // Generate unique key based on route path (e.g., /api/techniques or /api/workouts/winger)
    const cacheKey = `football_cache:${req.originalUrl || req.url}`;
    
    if (!cacheClient.isOpen) {
        return next(); // Skip cache safely if Redis is down
    }

    try {
        const cachedData = await cacheClient.get(cacheKey);
        if (cachedData) {
            return res.json(JSON.parse(cachedData)); // Return instantly
        }
        
        // Intercept response to save database data to cache before sending
        res.originalJson = res.json;
        res.json = (body) => {
            cacheClient.setEx(cacheKey, 3600, JSON.stringify(body)); // Cache for 1 hour
            res.originalJson(body);
        };
        next();
    } catch (err) {
        next();
    }
}

async function checkRateLimit(endpointType, maxRequests, windowSeconds = 60) {
    return async (req, res, next) => {
        if (!cacheClient || !cacheClient.isOpen) {
            return next(); // Safe fallback if Redis is down
        }

        const ip = req.ip || req.headers['x-forwarded-for'] || req.socket.remoteAddress;
        const key = `ratelimit:${ip}:${endpointType}`;
        const currentTime = Math.floor(Date.now() / 1000);

        try {
            const pipe = cacheClient.multi();
            pipe.zRemRangeByScore(key, 0, currentTime - windowSeconds);
            pipe.zAdd(key, { score: currentTime, value: currentTime.toString() + Math.random() });
            pipe.zCard(key);
            pipe.expire(key, windowSeconds);
            
            const results = await pipe.exec();
            const requestCount = results[2];

            if (requestCount > maxRequests) {
                const message = endpointType === 'auth' 
                    ? "Too many login attempts. Please try again in a minute."
                    : "Speed limit exceeded. Take a breath before your next drill!";
                return res.status(429).json({ error: message });
            }
            next();
        } catch (err) {
            console.error('Rate limiting error:', err);
            next();
        }
    };
}
